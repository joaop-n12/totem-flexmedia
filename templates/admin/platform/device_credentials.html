4. Exemplo de chamada (script no totem)

#!/bin/bash
# /opt/totem-flexmedia/heartbeat.sh — rodar via cron a cada minuto
set -euo pipefail
. /etc/totem/credenciais.env

BODY='{"app_version":"1.0.0","cpu_temp_c":48.3,"free_disk_mb":2300,"uptime_seconds":'$(awk '{print int($1)}' /proc/uptime)'}'
TS=$(date +%s)
SIG=$(printf '%s.%s' "$TS" "$BODY" | openssl dgst -sha256 -hmac "$TOTEM_API_KEY_SECRET" -hex | awk '{print $2}')

curl -fsS -X POST "$TOTEM_HEARTBEAT_URL" \
  -H "Content-Type: application/json" \
  -H "X-Totem-Key: $TOTEM_API_KEY_ID" \
  -H "X-Totem-Secret: $TOTEM_API_KEY_SECRET" \
  -H "X-Totem-Timestamp: $TS" \
  -H "X-Totem-Signature: $SIG" \
  --data "$BODY"
